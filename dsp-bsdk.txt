import hashlib
import csv
from tqdm import tqdm

# Sample dictionary
words = ["hello", "admin", "test123", "password123", "student"]

# Input password
password = input("Enter password: ")
hash_value = hashlib.sha256(password.encode()).hexdigest()

# Password strength
if len(password) < 6:
    strength = "Weak"
elif len(password) < 10:
    strength = "Medium"
else:
    strength = "Strong"

print("SHA256 Hash:", hash_value)
print("Password Strength:", strength)

# Dictionary attack simulation on your own input
found = False
for word in tqdm(words, desc="Testing words"):
    if word == password or hashlib.sha256(word.encode()).hexdigest() == hash_value:
        print("Password found:", word)
        found = True
        break

if not found:
    print("Password not found in dictionary")

# Export report
with open("report.csv", "w", newline="") as file:
    writer = csv.writer(file)
    writer.writerow(["Password", "SHA256 Hash", "Strength"])
    writer.writerow([password, hash_value, strength])

print("Report saved as report.csv")

    max_depth=3,
    random_state=0
)

    len(url),

from cryptography.fernet import Fernet

# Shared key for sender and receiver
key = Fernet.generate_key()
sender = Fernet(key)
receiver = Fernet(key)

# Sender writes a message
message = input("Sender: Enter message: ")

# Encrypt message
encrypted = sender.encrypt(message.encode())
print("Encrypted message sent to server:", encrypted)

# Server relays ciphertext only
print("Server: Forwarding encrypted message...")

# Receiver decrypts message
decrypted = receiver.decrypt(encrypted).decode()
print("Receiver received:", decrypted)

# @title Exp - 10
from cryptography.hazmat.primitives.asymmetric import rsa, padding
from cryptography.hazmat.primitives import hashes

# Generate keys
private_key = rsa.generate_private_key(
    public_exponent=65537, key_size=2048)
public_key = private_key.public_key()

# Digital Signature
message = b"Hello World"
signature = private_key.sign(
    message, padding.PKCS1v15(), hashes.SHA256())
print("Digital Signature Generated")

# Verify Signature
try:
    public_key.verify(
        signature, message, padding.PKCS1v15(), hashes.SHA256())
    print("Signature Verified")
except:
    print("Invalid Signature")

import pandas as pd

data = {
    "Age": [21, 22, 21, 22, 35, 36],
    "City": ["Mysore", "Mysore", "Delhi",
             "Delhi", "Mumbai", "Mumbai"],
    "Disease": ["Flu", "Cold", "Flu",
                "Cold", "Cancer", "Flu"]
}

df = pd.DataFrame(data)

print("Before Anonymization:")
print(df)

# Generalize age
df["Age"] = (df["Age"] // 10) * 10
df["Age"] = df["Age"].astype(str) + "s"

# Hide city
df["City"] = "*"

print("\nAfter Anonymization:")
print(df)

# Check 2-anonymity
groups = df.groupby(["Age", "City"]).size()
print("\nGroup Sizes:")
print(groups)

print("2-anonymous:", all(groups >= 2))
